                      {(msg.type === "text" || !msg.type) && (
                        <div className="flex flex-col gap-2">

                              onClick={(e) => { e.stopPropagation(); onGoToSupport(); }}
                              className="mt-1 px-3 py-1.5 rounded-lg bg-white/20 hover:bg-white/30 border border-white/20 text-white text-[13px] font-medium transition-colors w-fit flex items-center gap-2 shadow-sm"
                            >
                              <MessageCircle size={14} /> Go to Support
